-- Validate EMF reading request
function ToolValidator.ValidateEMFReading(player, position)
    -- Check if player has EMF tool equipped
    local character = player.Character
    if not character then return false end
    
    local emfTool = character:FindFirstChild("DetektorMEDOK")
    if not emfTool then return false end
    
    -- Check range (max 12 studs from player)
    local playerPos = character.HumanoidRootPart.Position
    local distance = (position - playerPos).Magnitude
    
    if distance > 12 then return false end
    
    return true
end

return ToolValidator

    Atmosphere density and fog increase as average team sanity drops
    
    Sanity 100-80: Light fog (Density 0.3)
    Sanity 79-50: Medium fog (Density 0.5)
    Sanity 49-20: Heavy fog (Density 0.7)
    Sanity <20: Extreme fog (Density 0.9)
]]

local Lighting = game:GetService("Lighting")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")

local atmosphere = Lighting:FindFirstChild("HorrorAtmosphere")
if not atmosphere then
    warn("[DynamicAtmosphere] HorrorAtmosphere not found")
    return
end

-- Update atmosphere based on sanity
local function updateAtmosphere(averageSanity)
    averageSanity = math.clamp(averageSanity, 0, 100)
    
    local targetDensity
    if averageSanity >= 80 then
        targetDensity = 0.3 -- Light fog
    elseif averageSanity >= 50 then
        targetDensity = 0.5 -- Medium fog
    elseif averageSanity >= 20 then
        targetDensity = 0.7 -- Heavy fog
    else
        targetDensity = 0.9 -- Extreme fog (hunt imminent)
    end
    
    -- Smooth transition
    local tween = TweenService:Create(
        atmosphere,
        TweenInfo.new(2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
        {Density = targetDensity}
    )
    tween:Play()
    
    print("[DynamicAtmosphere] Sanity:", averageSanity, "→ Density:", targetDensity)
end

-- Listen for team sanity updates
local remoteEvents = ReplicatedStorage:WaitForChild("RemoteEvents")
local teamSanityEvent = remoteEvents:FindFirstChild("TeamSanityUpdate")

if not teamSanityEvent then
    -- Create if doesn't exist
    teamSanityEvent = Instance.new("RemoteEvent")
    teamSanityEvent.Name = "TeamSanityUpdate"
    teamSanityEvent.Parent = remoteEvents
end

teamSanityEvent.OnClientEvent:Connect(function(averageSanity)
    updateAtmosphere(averageSanity)
end)

print("[DynamicAtmosphere] Initialized")

lua-- ServerScriptService/SanitySystem/Controller.lua
-- Add team sanity calculation and broadcast

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local teamSanityEvent = ReplicatedStorage.RemoteEvents.TeamSanityUpdate

local function broadcastTeamSanity(matchId)
    local match = _G.SystemRegistry:Get("MatchSystem"):GetMatch(matchId)
    if not match then return end
    
    local players = match.players
    if #players == 0 then return end
    
    local totalSanity = 0
    for _, playerId in ipairs(players) do
        local player = game.Players:GetPlayerByUserId(playerId)
        if player then
            local sanityData = match.sanityData[player.UserId]
            totalSanity = totalSanity + (sanityData and sanityData.current or 100)
        end
    end
    
    local averageSanity = totalSanity / #players
    
    -- Fire to all players in match
    for _, playerId in ipairs(players) do
        local player = game.Players:GetPlayerByUserId(playerId)
        if player then
            teamSanityEvent:FireClient(player, averageSanity)
        end
    end
end

-- Call this function whenever any player's sanity changes

lua-- ServerScriptService/RoyalPassSystem/Main.lua

local RoyalPassSystem = {}
RoyalPassSystem.__index = RoyalPassSystem

-- Royal Pass tier rewards
local REWARD_TIERS = {
    -- Level 1-100 rewards defined here
    -- Free tier: every 5 levels
    -- Premium tier: every level
}

function RoyalPassSystem.new()
    local self = setmetatable({}, RoyalPassSystem)
    self.playerData = {} -- {[userId] = {level, xp, premiumActive, claimedRewards}}
    return self
end

function RoyalPassSystem:Init()
    print("[RoyalPassSystem] Initializing...")
    
    -- Set up RemoteEvents
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local remoteEvents = ReplicatedStorage:FindFirstChild("RemoteEvents")
    
    local getRoyalPass = Instance.new("RemoteFunction")
    getRoyalPass.Name = "GetRoyalPass"
    getRoyalPass.Parent = remoteEvents
    getRoyalPass.OnServerInvoke = function(player)
        return self:GetPlayerData(player.UserId)
    end
    
    local claimRoyalPass = Instance.new("RemoteEvent")
    claimRoyalPass.Name = "ClaimRoyalPass"
    claimRoyalPass.Parent = remoteEvents
    claimRoyalPass.OnServerEvent:Connect(function(player)
        self:ClaimRewards(player.UserId)
    end)
    
    local royalPassUpdate = Instance.new("RemoteEvent")
    royalPassUpdate.Name = "RoyalPassUpdate"
    royalPassUpdate.Parent = remoteEvents
end

function RoyalPassSystem:Start()
    print("[RoyalPassSystem] Started")
end

function RoyalPassSystem:AddXP(userId, amount)
    local data = self:GetPlayerData(userId)
    data.xp = data.xp + amount
    
    -- Level up if needed
    while data.xp >= 1000 and data.level < 100 do
        data.xp = data.xp - 1000
        data.level = data.level + 1
        print("[RoyalPass] Player", userId, "reached level", data.level)
    end
    
    -- Broadcast update to client
    local player = game.Players:GetPlayerByUserId(userId)
    if player then
        local remoteEvents = game.ReplicatedStorage.RemoteEvents
        local royalPassUpdate = remoteEvents.RoyalPassUpdate
        royalPassUpdate:FireClient(player, data)
    end
end

function RoyalPassSystem:GetPlayerData(userId)
    if not self.playerData[userId] then
        -- Load from DataStore or create new
        self.playerData[userId] = {
            level = 1,
            xp = 0,
            premiumActive = false, -- Check if player purchased premium
            claimedRewards = {}
        }
    end
    return self.playerData[userId]
end

    -- Set transparency

    end

-- Update ghost state (called from state machine)
local function updateGhostState(ghostInstance, newState)
    local animator = ghostInstance.model:FindFirstChild("Humanoid"):FindFirstChild("Animator")
    if not animator then
        warn("[GhostSystem] Animator not found on ghost model")
        return
    end
    
    -- Load animations
    local animations = ghostInstance.model.Parent.Animations
    
    -- Stop all current animations
    for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
        track:Stop()
    end
    
    -- Play new animation based on state
    local animToPlay
    if newState == "Idle" then
        animToPlay = animations.Idle
    elseif newState == "Roaming" then
        animToPlay = animations.Walk or animations.Idle
    elseif newState == "Manifestation" then
        animToPlay = animations.Manifestation or animations.Idle
    elseif newState == "Hunting" then
        animToPlay = animations.Hunt or animations.Run
    end
    
    if animToPlay then
        local animTrack = animator:LoadAnimation(animToPlay)
        animTrack:Play()
    end
end

local ToolHandler = {}

-- EMF Reader usage
function ToolHandler.HandleEMFReader(player, position)
    local character = player.Character
    if not character then return end
    
    -- Validate player has tool
    local emfTool = character:FindFirstChild("DetektorMEDOK")
    if not emfTool then return end
    
    -- Check range
    local playerPos = character.HumanoidRootPart.Position
    local distance = (position - playerPos).Magnitude
    if distance > 12 then return end -- Max range: 12 studs
    
    -- Get current match
    local matchId = player:GetAttribute("CurrentMatch")
    if not matchId then return end
    
    local match = _G.SystemRegistry:Get("MatchSystem"):GetMatch(matchId)
    if not match then return end
    
    -- Check for ghost proximity (EMF reading)
    local ghostPos = match.ghostInstance and match.ghostInstance.model.PrimaryPart.Position
    if not ghostPos then
        return 0 -- No ghost, no EMF
    end
    
    local ghostDistance = (position - ghostPos).Magnitude
    
    -- EMF levels based on distance
    local emfLevel = 0
    if ghostDistance < 5 then
        emfLevel = 5 -- Very close
    elseif ghostDistance < 10 then
        emfLevel = 4
    elseif ghostDistance < 20 then
        emfLevel = 3
    elseif ghostDistance < 30 then
        emfLevel = 2
    elseif ghostDistance < 50 then
        emfLevel = 1
    end
    
    -- Fire to client for display update
    local remoteEvents = game.ReplicatedStorage.RemoteEvents
    local emfUpdateEvent = remoteEvents.EMFUpdate
    emfUpdateEvent:FireClient(player, emfLevel)
    
    -- Log evidence if EMF 5
    if emfLevel == 5 then
        local evidenceSystem = _G.SystemRegistry:Get("EvidenceSystem")
        evidenceSystem:RecordEvidence(matchId, player.UserId, "MEDOK")
    end
    
    return emfLevel
end

local activeParticles = 0
local MAX_PARTICLES = 200

function PerformanceOptimizer.EnableParticle(emitter)
    if activeParticles >= MAX_PARTICLES then
        warn("[PerformanceOptimizer] Max particles reached, skipping")
        return false
    end
    
    emitter.Enabled = true
    activeParticles = activeParticles + 1
    
    -- Auto-disable after duration
    task.delay(emitter.Lifetime.Max * 2, function()
        if emitter.Enabled then
            emitter.Enabled = false
            activeParticles = activeParticles - 1
        end
    end)
    
    return true
end

-- StreamingEnabled configuration
function PerformanceOptimizer.ConfigureStreaming()
    Workspace.StreamingEnabled = true
    Workspace.StreamingMinRadius = 128 -- Minimum radius always loaded
    Workspace.StreamingTargetRadius = 256 -- Target streaming radius
    
    print("[PerformanceOptimizer] StreamingEnabled configured")
end

-- Memory leak prevention (clean up old matches)
function PerformanceOptimizer.CleanupOldMatches()
    local activeMatches = Workspace:FindFirstChild("ActiveMatches")
    if not activeMatches then return end
    
    for _, matchFolder in ipairs(activeMatches:GetChildren()) do
        -- Check if match is completed (no players)
        if #matchFolder:GetChildren() == 0 then
            matchFolder:Destroy()
            print("[PerformanceOptimizer] Cleaned up old match:", matchFolder.Name)
        end
    end
end

-- Run cleanup every 60 seconds
task.spawn(function()
    while true do
        task.wait(60)
        PerformanceOptimizer.CleanupOldMatches()
    end
end)

PerformanceOptimizer.ConfigureStreaming()

print("[PerformanceOptimizer] Initialized")
